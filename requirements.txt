# Libraries the application needs to run.
# Each step adds its own libraries here, with a comment saying why.

# --- Step 0: the web application ---
fastapi                  # the web framework: turns Python functions into web endpoints
uvicorn[standard]        # the server program that runs the FastAPI application
pydantic                 # data validation: checks that data has the right shape
pydantic-settings        # reads configuration from environment variables and .env files

# --- Step 1: the database ---
sqlalchemy[asyncio]>=2.0 # talks to the database from Python; [asyncio] = non-blocking mode
asyncpg                  # the fast Postgres driver that SQLAlchemy uses underneath
alembic                  # database migrations: versioned, reviewable schema changes

# --- Step 2: the AI gateway ---
anthropic                # the official client for the Claude AI service

# --- Step 3: ingestion ---
httpx                    # talks to the job boards' public web interfaces
beautifulsoup4           # turns HTML job descriptions into clean readable text

# --- Step 4: resumes, embeddings, matching ---
openai                   # the embedding service client (meaning vectors)
pgvector                 # the vector column type for SQLAlchemy models
pypdf                    # extracts text from text-based PDF resumes
python-multipart         # lets FastAPI receive file uploads
